{{- /* Reads static/comics/ and returns a slice of dicts: name, dir, rel, url, title, author, description, direction, pages (count) A folder counts as a comic when it contains a comic.json holding the viewer metadata (title, author, description, direction, cover) — the page list is not read from it, it is counted from the images on disk via partials/lib/comic-pages.html. Drop a new folder with a comic.json into static/comics/ and the comics page fills itself in — one source of truth for the list and any counters. `url` is the standalone Comic Viewer embed for the comic, built by partials/lib/viewer-url.html. The list and single pages link to the Hugo page rather than to this, so it is only for direct-to-viewer links. */ -}} {{- $comics := slice -}} {{- $root := "static/comics" -}} {{- with (readDir $root) -}} {{- range . -}} {{- if .IsDir -}} {{- $dir := .Name -}} {{- $metaPath := printf "%s/%s/comic.json" $root $dir -}} {{- if fileExists $metaPath -}} {{- $m := dict -}} {{- with readFile $metaPath | transform.Unmarshal -}}{{- $m = . -}}{{- end -}} {{- /* Count pages from the images on disk, not from a hand-kept list in the manifest, so the card always matches what the viewer loads. */ -}} {{- $pages := partial "lib/comic-pages.html" $dir -}} {{- $title := default (replaceRE "[-_]" " " $dir) ($m.title) -}} {{- $comics = $comics | append (dict "name" $title "dir" $dir "rel" (printf "comics/%s" $dir) "url" (partial "lib/viewer-url.html" $dir) "title" $title "author" (default "" ($m.author)) "description" (default "" ($m.description)) "direction" (default "ltr" ($m.direction)) "pages" (len $pages)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $comics -}}
