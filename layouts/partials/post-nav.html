{{- /* Context: dict "pages" (page collection, ordered oldest → newest) "here" (the page you are on) Index idx-1 is one step earlier in that order, i.e. OLDER; idx+1 is NEWER. */ -}} {{- $pages := .pages -}} {{- $here := .here -}} {{- $idx := -1 -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $here.RelPermalink -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if gt $idx -1 -}} {{- $n := len $pages -}} {{- $older := "" -}}{{- $newer := "" -}} {{- if gt $idx 0 -}}{{- $older = index $pages (sub $idx 1) -}}{{- end -}} {{- if lt $idx (sub $n 1) -}}{{- $newer = index $pages (add $idx 1) -}}{{- end -}} {{- if or $older $newer }}
{{- with $older }} ← Older {{ .Title }} {{- end }} {{- with $newer }} Newer → {{ .Title }} {{- end }}
{{- end -}} {{- end -}}
